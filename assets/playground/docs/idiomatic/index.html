Idiomatic Karl

Karl favors expression flow and compact transforms. Prefer direct expressions over ceremony and helper wrappers unless reuse is real.

1. Style Rules

  • Keep bindings close to where values are used.
  • Prefer expression return style over multi-step temporary blocks.
  • Use object shorthand for readability: { id, email, }.
  • Reach for match + guards rather than long nested if ladders.

2. Concise Data Shaping

// compact object construction
let mkJSONHeaders = () -> map().set("Content-Type", "application/json")

let jsonResponse = (status, payload) -> {
    status,
    headers: mkJSONHeaders(),
    body: jsonEncode(payload),
}

let users = rows
    .filter(u -> u.active)
    .map(u -> { id: u.id, email: u.email, })

3. Recover and Defaults

// fail fast by default, recover only where fallback is meaningful
let cfg = jsonDecode(raw) ? { mode: "safe", retries: 3, }

// recover inline and continue expression flow
let quota = parseInt(env("MAX_RETRY") ? "3") ? { 3 }

4. Tasks and Concurrency

// spawn group + wait
let grouped = spawn {
    fetchUser("u1"),
    fetchUser("u2"),
}
let [u1, u2, ] = wait grouped

// race first result
let fastest = wait race {
    fetchFrom("a"),
    fetchFrom("b"),
}

// chain async transforms
let score = wait (& compute()).then(x -> x + 1).then(x -> x * 10)

5. Streaming with Channels

let ch = channel()

let producer = spawn (c -> {
    c.send("a")
    c.send("b")
    c.done()
})(ch)

let stream = for true with res = ch.recv(), acc = [] {
    let [v, done] = res
    if done { break acc }
    acc += [v]
    res = ch.recv()
} then acc

wait producer
log(stream)

6. Match + Guards

let classify = n -> match n {
    case _ if n < 0 -> "negative"
    case 0 -> "zero"
    case _ if n < 10 -> "small"
    case _ -> "large"
}

let msg = match event {
    case { kind: "email", payload: { id, subject, } } -> "mail:" + id + ":" + subject
    case { kind } -> "event:" + kind
    case _ -> "unknown"
}
