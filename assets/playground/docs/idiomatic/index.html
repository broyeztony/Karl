Idiomatic Karl

Karl shines when code stays expression-first, compact, and direct. Prefer a single clear expression over boilerplate wrappers. Keep abstractions only when reused enough to justify them.

1. Mindset

  • Think in values: every block returns a value, every branch returns a value.
  • Prefer local clarity over generic utility helpers.
  • Use spawn/race to communicate intent quickly.
  • Recover only where a safe fallback exists; fail fast elsewhere.
  • Keep data structure shape visible in-place with object shorthand.

2. Expression-First Helpers

Keep tiny helpers as single expressions. Avoid wrappers that just rename existing built-ins. 

// good: one-expression helpers
let mkJSONHeaders = () -> map().set("Content-Type", "application/json")
let normalizeEmail = s -> s.trim().toLower()

// avoid: pass-through aliases with no semantic value
// let msToRFC3339 = ms -> timeFormatRFC3339(ms)
// let rfc3339ToMs = txt -> timeParseRFC3339(txt)

3. Data Shaping + Shorthand

let userToDTO = u -> {
    id: u.id,
    email: u.email.toLower(),
    active: u.active,
}

let activeUsers = users
    .filter(u -> u.active)
    .map(userToDTO)

let result = {
    count: activeUsers.length,
    users: activeUsers,
}

let id = "u_42"
let email = "ana@karl.dev"
log({ id, email, }) // shorthand

4. Recover at Boundaries

Recover around IO/parsing boundaries, not around all business logic. 

// decode boundary
let cfg = jsonDecode(rawConfig) ? {
    log("invalid config:", error.message)
    { mode: "safe", retries: 3, }
}

// arithmetic boundary with explicit recover
let divide = (a, b) -> a / b
let value = divide(10, 0) ? {
    log("recover:", error.message)
    0
}

// parse boundary
let tickMs = parseInt(tickText) ? { 1000 }

5. Match + Guards

let classify = n -> match n {
    case _ if n < 0 -> "negative"
    case 0 -> "zero"
    case _ if n < 10 -> "small"
    case _ -> "large"
}

let route = event -> match event {
    case { kind: "email", payload: { user_id, subject, }, } -> {
        type: "notify",
        user_id,
        subject,
    }
    case { kind: "system", payload, } if payload.priority >= 10 -> {
        type: "alert",
        payload,
    }
    case { kind } -> { type: "ignore", kind, }
    case _ -> { type: "invalid", }
}

6. Collection Flows

let emails = [
    { id: "e1", score: 7, },
    { id: "e2", score: 2, },
    { id: "e3", score: 9, },
]

let important = emails
    .filter(e -> e.score >= 5)
    .sort((a, b) -> b.score - a.score)

let scoreSum = important
    .map(e -> e.score)
    .sum()

let firstCritical = important.find(e -> e.score >= 8)
log({ important, scoreSum, firstCritical, })

8. Tasks and spawn/race

let grouped = spawn {
    (() -> { sleep(80); { src: "a", score: 4, } })(),
    (() -> { sleep(120); { src: "b", score: 8, } })(),
}

let [a, b, ] = wait grouped

let fastest = wait race {
    (() -> { sleep(100); "slow" })(),
    (() -> { sleep(20); "fast" })(),
}

let computed = wait (& (() -> 20)()).then(x -> x + 1).then(x -> x * 2)
log({ a, b, fastest, computed, })

9. Channels Stream Pattern

let ch = buffered(64)

let producer = spawn (c -> {
    c.send({ id: "e1", score: 4, })
    c.send({ id: "e2", score: 9, })
    c.send({ id: "e3", score: 7, })
    c.done()
})(ch)

let consumed = for true with res = ch.recv(), acc = [] {
    let [msg, done] = res
    if done { break acc }
    if msg.score >= 5 {
        acc += [msg.id]
    }
    res = ch.recv()
} then acc

wait producer
log(consumed)

10. Runtime + Signals Pattern

// CLI-oriented pattern
let mode = match env("MODE") {
    case null -> "dev"
    case v -> v
}

let sigs = signalWatch(["SIGINT", "SIGTERM"])
let worker = spawn (() -> {
    for true with running = true {
        sleep(1000)
        log("tick", mode)
    } then ()
})()

let [sig, closed] = sigs.recv()
if !closed {
    log("shutdown:", sig)
}
worker.cancel()
let _ = (wait worker) ? {}

11. SQL + Time Pattern

let db = sqlOpen(dsn)
let nowMs = now()

let _ = sqlExec(db,
    "INSERT INTO users (id, email, last_email_check) VALUES ($1, $2, $3)",
    [userId, email, nowMs]
)

let row = sqlQueryOne(db,
    "SELECT id, email, last_email_check FROM users WHERE id = $1",
    [userId]
)

let view = match row {
    case null -> { found: false, }
    case r -> {
        found: true,
        id: r.id,
        email: r.email,
        checked_at: timeFormatRFC3339(r.last_email_check),
    }
}

log(view)
sqlClose(db)

12. Anti-Patterns to Avoid

  • One-off wrappers that only rename built-ins.
  • Introducing helper functions before confirming reuse.
  • Using if/else chains where match + guards is clearer.
  • Recovering every expression instead of boundary operations.
  • Verbose object construction when shorthand is sufficient.
// noisy
let mk = () -> {
    let h = map()
    h.set("Content-Type", "application/json")
    h
}

// idiomatic
let mk = () -> map().set("Content-Type", "application/json")
